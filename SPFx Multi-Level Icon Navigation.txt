```typescript
import { SPHttpClient, SPHttpClientResponse } from '@microsoft/sp-http';
import { INavigationItem } from '../models/INavigationItem';

export class NavigationService {

  private _spHttpClient: SPHttpClient;
  private _siteUrl: string;
  private _listName: string;

  constructor(spHttpClient: SPHttpClient, siteUrl: string, listName: string) {
    this._spHttpClient = spHttpClient;
    this._siteUrl = siteUrl;
    this._listName = listName;
  }

  public getNavigationItems(): Promise<INavigationItem[]> {

    const url: string =
      this._siteUrl
      + `/_api/web/lists/getbytitle('${this._listName}')/items`
      + `?$select=Id,Title,IconUrl,ParentItem/Id,ParentItem/Title,LinkUrl,SortOrder,IsActive,OpenInNewTab,Description`
      + `&$expand=ParentItem`
      + `&$filter=IsActive eq 1`
      + `&$orderby=SortOrder asc`
      + `&$top=500`;

    return this._spHttpClient.get(url, SPHttpClient.configurations.v1)
      .then((response: SPHttpClientResponse) => {
        if (!response.ok) {
          throw new Error('Failed to fetch navigation data: ' + response.statusText);
        }
        return response.json();
      })
      .then((data: any) => {
        const items: INavigationItem[] = [];

        data.value.forEach((item: any) => {
          items.push({
            Id: item.Id,
            Title: item.Title || '',
            IconUrl: item.IconUrl || '',
            ParentItemId: item.ParentItem ? item.ParentItem.Id : null,
            ParentItemTitle: item.ParentItem ? item.ParentItem.Title : '',
            LinkUrl: item.LinkUrl || '',
            SortOrder: item.SortOrder || 0,
            IsActive: item.IsActive !== false,
            OpenInNewTab: item.OpenInNewTab || false,
            Description: item.Description || '',
            children: [],
            isExpanded: false,
            level: 0
          });
        });

        return items;
      });
  }

  public buildTree(flatItems: INavigationItem[]): INavigationItem[] {

    const map: { [key: number]: INavigationItem } = {};
    flatItems.forEach((item: INavigationItem) => {
      item.children = [];
      item.isExpanded = false;
      map[item.Id] = item;
    });

    const roots: INavigationItem[] = [];

    flatItems.forEach((item: INavigationItem) => {
      if (item.ParentItemId !== null && map[item.ParentItemId]) {
        const parent: INavigationItem = map[item.ParentItemId];
        item.level = parent.level + 1;
        parent.children.push(item);
      } else {
        item.level = 0;
        roots.push(item);
      }
    });

    this._sortRecursive(roots);
    return roots;
  }

  private _sortRecursive(items: INavigationItem[]): void {
    items.sort((a, b) => a.SortOrder - b.SortOrder);
    items.forEach((item: INavigationItem) => {
      if (item.children.length > 0) {
        this._sortRecursive(item.children);
      }
    });
  }
}
```

```typescript
import { INavigationItem } from '../models/INavigationItem';

export class NavigationRenderer {

  private _container: HTMLElement;
  private _tree: INavigationItem[] = [];

  constructor(container: HTMLElement) {
    this._container = container;
  }

  // ══════════════════════════════════════
  //  MAIN RENDER
  // ══════════════════════════════════════

  public render(tree: INavigationItem[]): void {
    this._tree = tree;
    this._container.innerHTML = '';

    // Top-level icon grid
    const grid: HTMLElement = document.createElement('div');
    grid.className = 'iconNav-grid';

    tree.forEach((item: INavigationItem) => {
      const tile: HTMLElement = this._createIconTile(item);
      grid.appendChild(tile);
    });

    this._container.appendChild(grid);

    // Expandable sub-menu area (hidden initially)
    const submenuArea: HTMLElement = document.createElement('div');
    submenuArea.className = 'iconNav-submenuArea';
    submenuArea.id = 'iconNav-submenuArea';
    this._container.appendChild(submenuArea);
  }

  // ══════════════════════════════════════
  //  ICON TILE
  // ══════════════════════════════════════

  private _createIconTile(item: INavigationItem): HTMLElement {
    const tile: HTMLElement = document.createElement('div');
    tile.className = 'iconNav-tile';
    tile.setAttribute('data-id', item.Id.toString());
    tile.title = item.Description || item.Title;

    if (item.IconUrl) {
      const img: HTMLImageElement = document.createElement('img');
      img.src = item.IconUrl;
      img.alt = item.Title;
      img.className = 'iconNav-tileImg';
      img.onerror = (): void => {
        img.style.display = 'none';
        const fb: HTMLElement = document.createElement('div');
        fb.className = 'iconNav-tileFallback';
        fb.textContent = item.Title.charAt(0);
        tile.insertBefore(fb, tile.firstChild);
      };
      tile.appendChild(img);
    } else {
      const fb: HTMLElement = document.createElement('div');
      fb.className = 'iconNav-tileFallback';
      fb.textContent = item.Title.charAt(0);
      tile.appendChild(fb);
    }

    const label: HTMLElement = document.createElement('span');
    label.className = 'iconNav-tileLabel';
    label.textContent = item.Title;
    tile.appendChild(label);

    tile.addEventListener('click', () => {
      this._onTopLevelClick(item, tile);
    });

    return tile;
  }

  // ══════════════════════════════════════
  //  TOP-LEVEL CLICK HANDLER
  // ══════════════════════════════════════

  private _onTopLevelClick(item: INavigationItem, tileEl: HTMLElement): void {
    const submenuArea: HTMLElement = document.getElementById('iconNav-submenuArea') as HTMLElement;

    // If already expanded, collapse it
    if (item.isExpanded) {
      item.isExpanded = false;
      tileEl.classList.remove('iconNav-tileActive');
      submenuArea.innerHTML = '';
      submenuArea.style.display = 'none';
      return;
    }

    // Collapse all others
    this._tree.forEach((i: INavigationItem) => { i.isExpanded = false; });
    const allTiles: NodeListOf<Element> = document.querySelectorAll('.iconNav-tileActive');
    allTiles.forEach((el: Element) => { el.classList.remove('iconNav-tileActive'); });

    // Expand this one
    item.isExpanded = true;
    tileEl.classList.add('iconNav-tileActive');

    submenuArea.innerHTML = '';
    submenuArea.style.display = 'block';

    if (item.children.length > 0) {
      // Header
      const header: HTMLElement = document.createElement('div');
      header.className = 'iconNav-submenuHeader';
      header.textContent = item.Title;
      submenuArea.appendChild(header);

      // Sub-items list
      const list: HTMLElement = this._createSubmenuList(item.children, 1);
      submenuArea.appendChild(list);
    } else {
      const empty: HTMLElement = document.createElement('div');
      empty.className = 'iconNav-empty';
      empty.textContent = 'No sub-items configured.';
      submenuArea.appendChild(empty);
    }
  }

  // ══════════════════════════════════════
  //  RECURSIVE SUB-MENU BUILDER
  // ══════════════════════════════════════

  private _createSubmenuList(items: INavigationItem[], level: number): HTMLElement {
    const ul: HTMLElement = document.createElement('ul');
    ul.className = 'iconNav-submenuList iconNav-level' + level;

    items.forEach((item: INavigationItem) => {
      const li: HTMLElement = document.createElement('li');
      li.className = 'iconNav-submenuItem';

      const hasChildren: boolean = item.children.length > 0;
      const hasLink: boolean = item.LinkUrl.trim() !== '';

      const row: HTMLElement = document.createElement('div');
      row.className = 'iconNav-submenuRow';

      if (hasLink && !hasChildren) {
        // ── LINK ITEM: clicks navigate to URL ──
        const link: HTMLAnchorElement = document.createElement('a');
        link.href = item.LinkUrl;
        link.className = 'iconNav-link';
        link.textContent = item.Title;
        if (item.OpenInNewTab) {
          link.target = '_blank';
          link.rel = 'noopener noreferrer';
        }
        row.appendChild(link);
        li.appendChild(row);

      } else if (hasChildren) {
        // ── EXPANDABLE ITEM: clicks expand sub-sub-items ──
        const titleSpan: HTMLElement = document.createElement('span');
        titleSpan.className = 'iconNav-toggle';
        titleSpan.textContent = item.Title;
        row.appendChild(titleSpan);

        const chevron: HTMLElement = document.createElement('span');
        chevron.className = 'iconNav-chevron';
        chevron.innerHTML = '&#9654;';
        row.appendChild(chevron);

        const childWrap: HTMLElement = document.createElement('div');
        childWrap.className = 'iconNav-childWrap';
        childWrap.style.display = 'none';

        const childList: HTMLElement = this._createSubmenuList(item.children, level + 1);
        childWrap.appendChild(childList);

        row.addEventListener('click', (e: Event) => {
          e.stopPropagation();
          const isOpen: boolean = childWrap.style.display !== 'none';
          childWrap.style.display = isOpen ? 'none' : 'block';
          chevron.classList.toggle('iconNav-chevronOpen');
          row.classList.toggle('iconNav-rowExpanded');
        });

        li.appendChild(row);
        li.appendChild(childWrap);

      } else {
        // ── LABEL ONLY: no link, no children ──
        const span: HTMLElement = document.createElement('span');
        span.className = 'iconNav-label';
        span.textContent = item.Title;
        row.appendChild(span);
        li.appendChild(row);
      }

      ul.appendChild(li);
    });

    return ul;
  }

  // ══════════════════════════════════════
  //  LOADING & ERROR
  // ══════════════════════════════════════

  public renderLoading(): void {
    this._container.innerHTML = '<div class="iconNav-loading">Loading navigation...</div>';
  }

  public renderError(msg: string): void {
    this._container.innerHTML =
      '<div class="iconNav-error">'
      + '<span>⚠ ' + msg + '</span>'
      + '<button id="iconNav-retry" class="iconNav-retryBtn">Retry</button>'
      + '</div>';
  }
}
```

```typescript
import { Version } from '@microsoft/sp-core-library';
import { type IPropertyPaneConfiguration, PropertyPaneTextField } from '@microsoft/sp-property-pane';
import { BaseClientSideWebPart } from '@microsoft/sp-webpart-base';

import { NavigationService } from './services/NavigationService';
import { NavigationRenderer } from './components/NavigationRenderer';
import { IconNavigationStyles } from './IconNavigationStyles';
import { INavigationItem } from './models/INavigationItem';

export interface IIconNavigationWebPartProps {
  listName: string;
  sectionTitle: string;
  sectionSubtitle: string;
}

export default class IconNavigationWebPart extends BaseClientSideWebPart<IIconNavigationWebPartProps> {

  private _renderer!: NavigationRenderer;
  private _service!: NavigationService;

  public render(): void {

    // Inject global CSS
    IconNavigationStyles.inject();

    // Build shell HTML
    const title: string = this.properties.sectionTitle || '';
    const subtitle: string = this.properties.sectionSubtitle || '';

    this.domElement.innerHTML =
      '<div class="iconNav-wrapper">'
      + (title ? '<div style="font-size:20px;font-weight:700;color:#2c3e50;padding:20px 20px 0;">' + this._escapeHtml(title) + '</div>' : '')
      + (subtitle ? '<div style="font-size:13px;color:#7f8c8d;padding:4px 20px 10px;">' + this._escapeHtml(subtitle) + '</div>' : '')
      + '<div id="iconNav-container"></div>'
      + '</div>';

    // Initialize
    const container: HTMLElement = this.domElement.querySelector('#iconNav-container') as HTMLElement;
    this._renderer = new NavigationRenderer(container);
    this._service = new NavigationService(
      this.context.spHttpClient,
      this.context.pageContext.web.absoluteUrl,
      this.properties.listName || 'NavigationConfig'
    );

    this._loadNavigation();
  }

  private _loadNavigation(): void {
    this._renderer.renderLoading();

    this._service.getNavigationItems()
      .then((items: INavigationItem[]) => {
        const tree: INavigationItem[] = this._service.buildTree(items);
        this._renderer.render(tree);
      })
      .catch((error: Error) => {
        console.error('IconNavigation Error:', error);
        this._renderer.renderError('Failed to load navigation. Check list name and permissions.');

        const retryBtn: HTMLElement | null = document.getElementById('iconNav-retry');
        if (retryBtn) {
          retryBtn.addEventListener('click', () => { this._loadNavigation(); });
        }
      });
  }

  private _escapeHtml(text: string): string {
    const div: HTMLElement = document.createElement('div');
    div.appendChild(document.createTextNode(text));
    return div.innerHTML;
  }
```

## PHASE 9: ONGOING MANAGEMENT (NO CODE NEEDED)

Everything is managed from the **NavigationConfig** list:

| What you want to do               | What to do in the list                                    |
|------------------------------------|-----------------------------------------------------------|
| Add new top-level icon             | New item, ParentItem=blank, set IconUrl & SortOrder       |
| Add sub-item under an icon         | New item, ParentItem=parent title                         |
| Add sub-sub-item                   | New item, ParentItem=the sub-item title                   |
| Make item a clickable link         | Fill in LinkUrl                                           |
| Make item expandable               | Leave LinkUrl blank, add children pointing to it          |
| Change display order               | Edit SortOrder (lower number = first)                     |
| Hide item temporarily              | Set IsActive = No                                         |
| Show hidden item                   | Set IsActive = Yes                                        |
| Change icon image                  | Update IconUrl path                                       |
| Rename item                        | Edit Title                                                |
| Delete permanently                 | Delete the list item                                      |
| Open link in new tab               | Set OpenInNewTab = Yes                                    |

## TROUBLESHOOTING

| Problem                              | Solution                                              |
|--------------------------------------|-------------------------------------------------------|
| "No gulpfile found"                  | EXPECTED in SPFx 1.22! Use `npm run build` instead    |
| npm run build fails with errors      | Check import paths match file names exactly            |
| Icons not loading                    | Check IconUrl paths — upload icons to SiteAssets       |
| "Failed to load navigation"          | Check list name is exactly "NavigationConfig"          |
| Sub-items not appearing              | Check ParentItem lookup is set correctly               |
| Web part not found when adding       | Make sure .sppkg is deployed and trusted               |
| Styles not showing                   | Hard refresh browser: Ctrl+Shift+R                    |
| Property pane not opening            | Click pencil/edit icon on the web part                 |
