# Restrict Archive files on SINGLE LIBRARY - Test Version
# Keep only NXT-CBS Directors

param(
    [Parameter(Mandatory=$true)]
    [string]$SiteUrl,
    
    [Parameter(Mandatory=$true)]
    [string]$LibraryName,  # Specify the library to test on
    
    [Parameter(Mandatory=$false)]
    [string]$AllowedGroupName = "NXT-CBS Directors"
)

# Connect to SharePoint
Write-Host "Connecting to SharePoint: $SiteUrl" -ForegroundColor Green
Connect-PnPOnline -Url $SiteUrl -Interactive

# Get the allowed group (NXT-CBS Directors)
$allowedGroup = Get-PnPGroup -Identity $AllowedGroupName -ErrorAction SilentlyContinue

if ($null -eq $allowedGroup) {
    Write-Host "ERROR: Group '$AllowedGroupName' not found" -ForegroundColor Red
    Write-Host "Available groups:" -ForegroundColor Yellow
    Get-PnPGroup | Select-Object Title
    exit
}

Write-Host "Target Group: $($allowedGroup.Title) (ID: $($allowedGroup.Id))" -ForegroundColor Green

# Get the specific library
$lib = Get-PnPList -Identity $LibraryName -ErrorAction SilentlyContinue

if ($null -eq $lib) {
    Write-Host "ERROR: Library '$LibraryName' not found" -ForegroundColor Red
    Write-Host "Available libraries:" -ForegroundColor Yellow
    Get-PnPList | Where-Object {$_.BaseType -eq "DocumentLibrary"} | Select-Object Title
    exit
}

Write-Host "Testing on Library: $($lib.Title)" -ForegroundColor Green
Write-Host "Starting permission restriction process...`n" -ForegroundColor Cyan

$processedCount = 0
$errorCount = 0

# Get all items with Archive content type in this library
$archiveItems = Get-PnPListItem -List $lib -PageSize 5000 | Where-Object {
    $_.FieldValues.ContentType -eq "Archive"
}

if ($archiveItems.Count -eq 0) {
    Write-Host "⚠ No archived items found in $($lib.Title)" -ForegroundColor Yellow
    exit
}

Write-Host "═══════════════════════════════════════" -ForegroundColor Cyan
Write-Host "Library: $($lib.Title)" -ForegroundColor Cyan
Write-Host "═══════════════════════════════════════" -ForegroundColor Cyan
Write-Host "Found $($archiveItems.Count) archived items`n"

foreach ($item in $archiveItems) {
    $itemName = $item.FieldValues.FileLeafRef
    
    try {
        # Step 1: Break inheritance (make unique)
        Set-PnPListItemPermission -List $lib -Identity $item.Id -InheritPermissions $false -ErrorAction Stop
        Write-Host "  📋 $itemName" -ForegroundColor Yellow
        Write-Host "     ├─ Breaking inheritance..." -ForegroundColor Gray
        
        # Step 2: Get all current permissions
        $perms = Get-PnPListItemPermission -List $lib -Identity $item.Id
        
        # Step 3: Remove all permissions except the target group
        Write-Host "     ├─ Removing existing groups..." -ForegroundColor Gray
        foreach ($perm in $perms) {
            # Skip if it's the group we want to keep
            if ($perm.PrincipalId -ne $allowedGroup.Id) {
                Remove-PnPListItemPermission -List $lib -Identity $item.Id -PrincipalId $perm.PrincipalId -Force
                Write-Host "        └─ Removed: $($perm.Title)" -ForegroundColor Red
            }
        }
        
        # Step 4: Grant permission to NXT-CBS Directors (if not already granted)
        $groupHasAccess = $perms | Where-Object {$_.PrincipalId -eq $allowedGroup.Id}
        
        if ($null -eq $groupHasAccess) {
            Set-PnPListItemPermission -List $lib -Identity $item.Id -PrincipalId $allowedGroup.Id -AddRole "Edit" -ErrorAction Stop
            Write-Host "     └─ ✓ Granted Edit permission to: $($allowedGroup.Title)" -ForegroundColor Green
        } else {
            Write-Host "     └─ ✓ $($allowedGroup.Title) already has permission" -ForegroundColor Green
        }
        
        $processedCount++
        
    } catch {
        Write-Host "     └─ ✗ ERROR: $($_.Exception.Message)" -ForegroundColor Red
        $errorCount++
    }
}

# Summary
Write-Host ""
Write-Host "════════════════════════════════════════" -ForegroundColor Green
Write-Host "TEST COMPLETE" -ForegroundColor Green
Write-Host "════════════════════════════════════════" -ForegroundColor Green
Write-Host "✓ Files Processed: $processedCount" -ForegroundColor Green
Write-Host "✗ Errors: $errorCount" -ForegroundColor $(if ($errorCount -eq 0) {"Green"} else {"Red"})
Write-Host "════════════════════════════════════════`n" -ForegroundColor Green

if ($errorCount -eq 0 -and $processedCount -gt 0) {
    Write-Host "✓ Test successful! You can now run on all libraries." -ForegroundColor Green
}

)

try {
    Write-Host "Connecting to SharePoint..." -ForegroundColor Green

        foreach ($field in $item.FieldValues.Keys) {
            if ($field -like "*Document*Type*") {
                if ($item.FieldValues[$field] -eq "Archive") {
                    $archiveItems += $item

        }

        $itemId = $item.Id

            Write-Host "📋 $itemName" -ForegroundColor Yellow
            
            # Get current permissions
            Write-Host "   ├─ Getting current permissions..." -ForegroundColor Gray

            if ($null -eq $perms) {
                Write-Host "   ├─ No permissions found, skipping..." -ForegroundColor Yellow
                continue
            }
            
            # Remove each permission EXCEPT the allowed group
            Write-Host "   ├─ Removing existing permissions..." -ForegroundColor Gray

                        Remove-PnPListItemPermission -List $lib -Identity $itemId -PrincipalId $perm.PrincipalId -Force -ErrorAction SilentlyContinue

            if ($null -eq $groupExists) {
                Set-PnPListItemPermission -List $lib -Identity $itemId -PrincipalId $allowedGroup.Id -AddRole "Edit" -ErrorAction Stop
                Write-Host "   └─ ✓ Permission granted" -ForegroundColor Green
            } else {
                Write-Host "   └─ ✓ Permission already exists" -ForegroundColor Green
            }
            
            Write-Host ""

            Write-Host "   └─ ✗ ERROR: $($_.Exception.Message)" -ForegroundColor Red
            Write-Host ""
            $errorCount++
        }
    }

    Write-Host "════════════════════════════════════" -ForegroundColor Green
    Write-Host "COMPLETE" -ForegroundColor Green

# Find and restrict Archive items - REST API VERSION
param(
    [Parameter(Mandatory=$true)]
    [string]$SiteUrl,
    
    [Parameter(Mandatory=$true)]
    [string]$LibraryName,
    
    [Parameter(Mandatory=$false)]
    [string]$AllowedGroupName = "NXT-CBS Directors"
)

try {
    Write-Host "Connecting to SharePoint..." -ForegroundColor Green
    Connect-PnPOnline -Url $SiteUrl -Interactive

    # Get admin group
    $allowedGroup = Get-PnPGroup -Identity $AllowedGroupName -ErrorAction SilentlyContinue
    if ($null -eq $allowedGroup) {
        Write-Host "ERROR: Group '$AllowedGroupName' not found" -ForegroundColor Red
        return
    }

    Write-Host "Admin Group: $($allowedGroup.Title) (ID: $($allowedGroup.Id))`n" -ForegroundColor Green

    # Get the library
    $lib = Get-PnPList -Identity $LibraryName -ErrorAction SilentlyContinue
    if ($null -eq $lib) {
        Write-Host "ERROR: Library '$LibraryName' not found" -ForegroundColor Red
        return
    }

    Write-Host "Library: $($lib.Title)" -ForegroundColor Cyan
    Write-Host "═════════════════════════════════════`n" -ForegroundColor Cyan

    # Get all items
    $allItems = Get-PnPListItem -List $lib -PageSize 5000

    # Find Archive items
    $archiveItems = @()
    
    foreach ($item in $allItems) {
        foreach ($field in $item.FieldValues.Keys) {
            if ($field -like "*Document*Type*") {
                if ($item.FieldValues[$field] -eq "Archive") {
                    $archiveItems += $item
                }
            }
        }
    }

    if ($archiveItems.Count -eq 0) {
        Write-Host "⚠ No Archive items found" -ForegroundColor Yellow
        return
    }

    Write-Host "Found $($archiveItems.Count) Archive items`n" -ForegroundColor Green

    $processedCount = 0
    $errorCount = 0
    $listId = $lib.Id.Guid

    foreach ($item in $archiveItems) {
        $itemName = $item.FieldValues.FileLeafRef
        $itemId = $item.Id
        
        try {
            Write-Host "📋 $itemName" -ForegroundColor Yellow
            
            # Step 1: Break inheritance
            Write-Host "   ├─ Breaking inheritance..." -ForegroundColor Gray
            $breakUrl = "$SiteUrl/_api/web/lists('$listId')/items($itemId)/breakroleinheritance(true)"
            
            try {
                Invoke-PnPSPRestMethod -Url $breakUrl -Method POST -ErrorAction SilentlyContinue | Out-Null
                Write-Host "   │  └─ ✓ Inheritance broken" -ForegroundColor Green
            } catch {
                Write-Host "   │  └─ ✓ Inheritance already broken" -ForegroundColor Gray
            }
            
            Start-Sleep -Milliseconds 300
            
            # Step 2: Get role assignments
            Write-Host "   ├─ Fetching role assignments..." -ForegroundColor Gray
            $rolesUrl = "$SiteUrl/_api/web/lists('$listId')/items($itemId)/roleassignments"
            $roles = Invoke-PnPSPRestMethod -Url $rolesUrl -Method GET
            
            if ($roles.value.Count -eq 0) {
                Write-Host "   ├─ No role assignments found" -ForegroundColor Yellow
                continue
            }
            
            # Step 3: Remove roles except admin group
            Write-Host "   ├─ Removing non-admin roles..." -ForegroundColor Gray
            $removed = 0
            $kept = 0
            
            foreach ($role in $roles.value) {
                $principalId = $role.PrincipalId
                $principalName = $role.Principal.Title
                
                if ($principalId -ne $allowedGroup.Id) {
                    try {
                        $deleteUrl = "$SiteUrl/_api/web/lists('$listId')/items($itemId)/roleassignments/getbyprincipalid($principalId)"
                        Invoke-PnPSPRestMethod -Url $deleteUrl -Method DELETE -ErrorAction Stop | Out-Null
                        Write-Host "      └─ ✓ Removed: $principalName" -ForegroundColor Red
                        $removed++
                    } catch {
                        Write-Host "      └─ ⚠ Could not remove: $principalName" -ForegroundColor Yellow
                    }
                } else {
                    Write-Host "      └─ ✓ Kept: $principalName" -ForegroundColor Green
                    $kept++
                }
            }
            
            Write-Host "   └─ ✓ Summary: Removed $removed, Kept $kept" -ForegroundColor Green
            Write-Host ""
            $processedCount++
            
        } catch {
            Write-Host "   └─ ✗ ERROR: $($_.Exception.Message)" -ForegroundColor Red
            Write-Host ""
            $errorCount++
        }
    }

    # Summary
    Write-Host "════════════════════════════════════" -ForegroundColor Green
    Write-Host "COMPLETE" -ForegroundColor Green
    Write-Host "════════════════════════════════════" -ForegroundColor Green
    Write-Host "✓ Processed: $processedCount" -ForegroundColor Green
    Write-Host "✗ Errors: $errorCount" -ForegroundColor $(if ($errorCount -eq 0) {"Green"} else {"Red"})
    Write-Host "════════════════════════════════════`n" -ForegroundColor Green
    
    Write-Host "RESULTS:" -ForegroundColor Cyan
    Write-Host "✓ Archive items now have unique permissions" -ForegroundColor Green
    Write-Host "✓ Only NXT-CBS Directors have access" -ForegroundColor Green
    Write-Host "✓ All other groups have been removed" -ForegroundColor Green

} catch {
    Write-Host "SCRIPT ERROR: $($_.Exception.Message)" -ForegroundColor Red
}

Read-Host "Press Enter to close"

# Restrict Archive items - CSOM Approach (Admin)
param(
    [Parameter(Mandatory=$true)]
    [string]$SiteUrl,
    
    [Parameter(Mandatory=$true)]
    [string]$LibraryName,
    
    [Parameter(Mandatory=$false)]
    [string]$AllowedGroupName = "NXT-CBS Directors"
)

try {
    # Load SharePoint CSOM assemblies
    Write-Host "Loading SharePoint assemblies..." -ForegroundColor Green
    Add-Type -Path "C:\Program Files\Common Files\Microsoft Shared\Web Server Extensions\16\ISAPI\Microsoft.SharePoint.Client.dll"
    Add-Type -Path "C:\Program Files\Common Files\Microsoft Shared\Web Server Extensions\16\ISAPI\Microsoft.SharePoint.Client.Runtime.dll"
    
    # Connect with credentials
    Write-Host "Connecting to SharePoint..." -ForegroundColor Green
    $credentials = Get-Credential
    $context = New-Object Microsoft.SharePoint.Client.ClientContext($SiteUrl)
    $context.Credentials = $credentials
    
    # Get web and list
    $web = $context.Web
    $list = $web.Lists.GetByTitle($LibraryName)
    $context.Load($web)
    $context.Load($list)
    $context.ExecuteQuery()
    
    Write-Host "Connected to: $($web.Title)" -ForegroundColor Green
    Write-Host "Library: $($list.Title)`n" -ForegroundColor Cyan
    
    # Get allowed group
    $groups = $web.SiteGroups
    $context.Load($groups)
    $context.ExecuteQuery()
    
    $allowedGroup = $groups | Where-Object {$_.Title -eq $AllowedGroupName}
    if ($null -eq $allowedGroup) {
        Write-Host "ERROR: Group '$AllowedGroupName' not found" -ForegroundColor Red
        return
    }
    
    Write-Host "Admin Group: $($allowedGroup.Title)`n" -ForegroundColor Green
    
    # Get all items with Archive document type
    $query = New-Object Microsoft.SharePoint.Client.CamlQuery
    $query.ViewXml = "<View><Query><Where><Eq><FieldRef Name='Document_x0020_Type'/><Value Type='Text'>Archive</Value></Eq></Where></Query></View>"
    
    $items = $list.GetItems($query)
    $context.Load($items)
    $context.ExecuteQuery()
    
    if ($items.Count -eq 0) {
        Write-Host "⚠ No Archive items found" -ForegroundColor Yellow
        return
    }
    
    Write-Host "Found $($items.Count) Archive items`n" -ForegroundColor Green
    
    $processedCount = 0
    $errorCount = 0
    
    foreach ($item in $items) {
        $itemName = $item["FileLeafRef"]
        
        try {
            Write-Host "📋 $itemName" -ForegroundColor Yellow
            
            # Break inheritance
            Write-Host "   ├─ Breaking inheritance..." -ForegroundColor Gray
            $item.BreakRoleInheritance($true)
            
            # Get role assignments
            $roleAssignments = $item.RoleAssignments
            $context.Load($roleAssignments)
            $context.ExecuteQuery()
            
            Write-Host "   ├─ Removing non-admin groups..." -ForegroundColor Gray
            
            # Delete role assignments for non-admin groups
            $toRemove = @()
            foreach ($assignment in $roleAssignments) {
                $principal = $assignment.Member
                $context.Load($principal)
                $context.ExecuteQuery()
                
                if ($assignment.PrincipalId -ne $allowedGroup.Id) {
                    $toRemove += $assignment
                    Write-Host "      └─ Marked for removal: $($principal.Title)" -ForegroundColor Red
                }
            }
            
            # Remove the marked assignments
            foreach ($assignment in $toRemove) {
                $roleAssignments.Remove($assignment)
            }
            
            # Add permission for allowed group
            Write-Host "   ├─ Adding Edit permission to $($allowedGroup.Title)..." -ForegroundColor Gray
            
            # Get Edit role
            $roleDefinitions = $web.RoleDefinitions
            $context.Load($roleDefinitions)
            $context.ExecuteQuery()
            
            $editRole = $roleDefinitions | Where-Object {$_.Name -eq "Edit"}
            
            # Create new role assignment
            $roleDefBindingCollection = New-Object Microsoft.SharePoint.Client.RoleDefinitionBindingCollection($context)
            $roleDefBindingCollection.Add($editRole)
            
            $roleAssignments.Add($allowedGroup, $roleDefBindingCollection)
            
            # Execute all changes
            $context.ExecuteQuery()
            
            Write-Host "   └─ ✓ Permissions updated" -ForegroundColor Green
            Write-Host ""
            $processedCount++
            
        } catch {
            Write-Host "   └─ ✗ ERROR: $($_.Exception.Message)" -ForegroundColor Red
            Write-Host ""
            $errorCount++
        }
    }
    
    # Summary
    Write-Host "════════════════════════════════════" -ForegroundColor Green
    Write-Host "COMPLETE" -ForegroundColor Green
    Write-Host "════════════════════════════════════" -ForegroundColor Green
    Write-Host "✓ Processed: $processedCount" -ForegroundColor Green
    Write-Host "✗ Errors: $errorCount" -ForegroundColor $(if ($errorCount -eq 0) {"Green"} else {"Red"})
    Write-Host "════════════════════════════════════`n" -ForegroundColor Green

} catch {
    Write-Host "ERROR: $($_.Exception.Message)" -ForegroundColor Red
    Write-Host "Stack Trace: $($_.ScriptStackTrace)" -ForegroundColor Red
}

Read-Host "Press Enter to close"

            $breakUrl = "$SiteUrl/_api/web/lists('$listId')/items($itemId)/breakroleinheritance(false)"
            Invoke-PnPSPRestMethod -Url $breakUrl -Method POST -ErrorAction Stop | Out-Null
            
            Start-Sleep -Milliseconds 200
            
            # Add Edit permission for allowed group
            $addPermUrl = "$SiteUrl/_api/web/lists('$listId')/items($itemId)/roleassignments/addroleassignment(principalid=$($allowedGroup.Id),roledefid=6)"
            Invoke-PnPSPRestMethod -Url $addPermUrl -Method POST -ErrorAction Stop | Out-Null
            
            $processedCount++
            
            # Show progress every 10 items
            if ($processedCount % 10 -eq 0) {
                Write-Host "✓ Processed: $processedCount / $($archiveItems.Count)" -ForegroundColor Green
            }
            
        } catch {
            Write-Host "✗ Error on $itemName : $($_.Exception.Message)" -ForegroundColor Red
            $errorCount++
        }
    }

    # Summary
    Write-Host "`n════════════════════════════════════" -ForegroundColor Green
    Write-Host "COMPLETE" -ForegroundColor Green
    Write-Host "════════════════════════════════════" -ForegroundColor Green
    Write-Host "✓ Processed: $processedCount" -ForegroundColor Green
    Write-Host "✗ Errors: $errorCount" -ForegroundColor $(if ($errorCount -eq 0) {"Green"} else {"Red"})
    Write-Host "════════════════════════════════════`n" -ForegroundColor Green
    
    Write-Host "RESULTS:" -ForegroundColor Cyan
    Write-Host "✓ All Archive items now have UNIQUE permissions" -ForegroundColor Green
    Write-Host "✓ All inherited permissions CLEARED" -ForegroundColor Green
    Write-Host "✓ Only NXT-CBS Directors have Edit access" -ForegroundColor Green

} catch {
    Write-Host "SCRIPT ERROR: $($_.Exception.Message)" -ForegroundColor Red
}
